cmake_minimum_required(VERSION 3.16)
project(mkauto_host_tests C)
enable_testing()

set(MK_IDF_PATH "$ENV{IDF_PATH}" CACHE PATH "ESP-IDF installation providing Unity")
set(mk_unity "${MK_IDF_PATH}/components/unity/unity/src")
if(NOT EXISTS "${mk_unity}/unity.c")
    message(FATAL_ERROR "Set MK_IDF_PATH to ESP-IDF 5.5.5")
endif()

foreach(mk_state IN ITEMS unknown false true)
    set(mk_target "test_build_info_${mk_state}")
    add_executable(${mk_target} main.c ../test_build_info.c ../../src/mk_build_info.c "${mk_unity}/unity.c")
    target_include_directories(${mk_target} PRIVATE stubs ../../inc "${mk_unity}")
    if(mk_state STREQUAL "unknown")
        set(mk_commit "unknown")
    else()
        set(mk_commit "0123456789ab")
    endif()
    target_compile_definitions(${mk_target} PRIVATE MK_BUILD_COMMIT="${mk_commit}" MK_BUILD_DIRTY="${mk_state}")
    set_target_properties(${mk_target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${mk_target} PRIVATE -Wall -Wextra -Werror -Wformat=2 -Wshadow)
    add_test(NAME ${mk_target} COMMAND ${mk_target})
endforeach()

add_executable(test_telemetry main_telemetry.c ../test_telemetry.c ../../lib/telemetry/src/mk_telemetry.c "${mk_unity}/unity.c")
target_include_directories(test_telemetry PRIVATE ../../lib/telemetry/inc "${mk_unity}")
set_target_properties(test_telemetry PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(test_telemetry PRIVATE -Wall -Wextra -Werror -Wformat=2 -Wshadow)
add_test(NAME test_telemetry COMMAND test_telemetry)

add_executable(test_gear main_gear.c ../test_gear.c ../../lib/gear/src/mk_gear.c "${mk_unity}/unity.c")
target_include_directories(test_gear PRIVATE ../../lib/gear/inc "${mk_unity}")
set_target_properties(test_gear PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(test_gear PRIVATE -Wall -Wextra -Werror -Wformat=2 -Wshadow)
target_link_libraries(test_gear PRIVATE m)
add_test(NAME test_gear COMMAND test_gear)
